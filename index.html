                const dateStr = `${year}-${(month+1).toString().padStart(2,'0')}-${d.toString().padStart(2,'0')}`;

                const cell = document.createElement('div'); const isToday = dateStr === todayStr;

                cell.className = `day-cell ${isToday ? 'today' : ''}`; cell.innerHTML = `<div class="day-num">${d}</div>`;

                if (chore.mode === 'single') {

                    const uid = rData[dateStr]; const area = document.createElement('div'); area.className = 'single-target-area';

                    if(uid) { const u = users.find(x=>x.id===uid); if(u) area.innerHTML = `<div class="stamp-avatar">${u.icon}</div><div class="locked-indicator">🔒</div>`; }

                    area.onclick = () => window.handleStamp(dateStr, null, uid); cell.appendChild(area);

                } else {

                    const dayObj = rData[dateStr] || {};

                    cell.appendChild(createSlot(0, dayObj[0], dateStr, '午')); cell.appendChild(createSlot(1, dayObj[1], dateStr, '晚'));

                }

                grid.appendChild(cell);

            }

            renderStats(year, month);

        }


        function createSlot(idx, uid, dStr, label) {

            const slot = document.createElement('div'); slot.className = 'split-slot';

            slot.onclick = (e) => { e.stopPropagation(); window.handleStamp(dStr, idx, uid); };

            slot.innerHTML = `<div class="slot-label">${label}</div>`;

            if(uid) { const u = users.find(x=>x.id===uid); if(u) slot.innerHTML += `<div class="stamp-avatar">${u.icon}</div><div class="locked-indicator">🔒</div>`; }

            return slot;

        }


        function renderStats(year, month) {

            const div = document.getElementById('statsGrid'); div.innerHTML = ''; if(!chores.length) return;

            const chore = chores.find(c=>c.id===currentChoreId); const rData = records[currentChoreId] || {};

            const counts = {}; users.forEach(u => counts[u.id] = 0);

            Object.keys(rData).forEach(d => {

                if(new Date(d).getFullYear() === year && new Date(d).getMonth() === month) {

                    const val = rData[d];

                    if(chore.mode === 'single') { if(counts[val] !== undefined) counts[val]++; } 

                    else { if(val[0] && counts[val[0]] !== undefined) counts[val[0]]++; if(val[1] && counts[val[1]] !== undefined) counts[val[1]]++; }

                }

            });

            users.forEach(u => { if(counts[u.id] > 0) div.innerHTML += `<div class="stat-item"><div style="font-size:20px">${u.icon}</div><div class="stat-count">${counts[u.id]}</div></div>`; });

            if(div.innerHTML === '') div.innerHTML = '<div style="padding:10px;color:#ccc">本月尚無紀錄</div>';

        }


        window.handleStamp = (dateStr, slotIndex, existingUid) => {

            if(!users.length) return alert("請先新增成員");

            if (!existingUid) { if(navigator.vibrate) navigator.vibrate(50); window.updateRecordInCloud(currentChoreId, dateStr, slotIndex, currentUserId); } 

            else { const u = users.find(x => x.id === existingUid); if(!u) { window.updateRecordInCloud(currentChoreId, dateStr, slotIndex, null); return; } pendingAction = { type: 'deleteRecord', correctPin: u.pin, dateStr, slotIndex, choreId: currentChoreId }; window.openPinModal(u, "輸入密碼以移除紀錄："); }

        };

        window.openUserModal = (userId) => {

            editingId = userId; const user = userId ? users.find(u=>u.id===userId) : null;

            document.getElementById('userModalTitle').innerText = user ? "編輯成員 (改名)" : "新增成員";

            document.getElementById('userNameInput').value = user ? user.name : "";

            document.getElementById('userPinSection').style.display = user ? 'none' : 'block';

            document.getElementById('userPinInput').value = ""; tempIcon = user ? user.icon : avatars[0];

            renderIconGrid('userIconGrid', avatars); document.getElementById('userModal').style.display = 'flex';

        };

        window.confirmSaveUser = () => {

            const name = document.getElementById('userNameInput').value.trim(); if(!name) return alert("請輸入名字");

            if(editingId) { const u = users.find(x => x.id === editingId); window.saveUserToCloud({ ...u, name, icon: tempIcon }); } 

            else { const pin = document.getElementById('userPinInput').value.trim(); if(!pin || pin.length !== 3) return alert("請設定3位數密碼"); const newId = 'u' + Date.now(); window.saveUserToCloud({ id: newId, name, icon: tempIcon, pin }); currentUserId = newId; } window.closeModal('userModal');

        };

        window.requestDeleteUser = (userId) => { const user = users.find(u=>u.id===userId); pendingAction = { type: 'deleteUser', correctPin: user.pin, userId }; window.openPinModal(user, `輸入 ${user.name} 的密碼以確認刪除：`); };

        window.openChoreModal = (choreId) => {

            editingId = choreId; const chore = choreId ? chores.find(c=>c.id===choreId) : null;

            document.getElementById('choreModalTitle').innerText = chore ? "編輯任務" : "新增任務";

            document.getElementById('choreNameInput').value = chore ? chore.name : "";

            const modeDiv = document.getElementById('choreModeSelect'); if(chore) { modeDiv.style.display = 'none'; tempMode = chore.mode; } else { modeDiv.style.display = 'flex'; window.selectChoreMode('single'); }

            tempIcon = chore ? chore.icon : choreIcons[0]; renderIconGrid('choreIconGrid', choreIcons); document.getElementById('choreModal').style.display = 'flex';

        };

        window.confirmSaveChore = () => {

            const name = document.getElementById('choreNameInput').value.trim(); if(!name) return alert("請輸入名稱");

            if(editingId) { const c = chores.find(x => x.id === editingId); window.saveChoreToCloud({ ...c, name, icon: tempIcon }); } else { const newId = 'c' + Date.now(); window.saveChoreToCloud({ id: newId, name, icon: tempIcon, mode: tempMode }); currentChoreId = newId; } window.closeModal('choreModal');

        };

        window.confirmDeleteChore = (choreId) => { const c = chores.find(x=>x.id===choreId); if(confirm(`確定要刪除「${c.name}」嗎？這會一併刪除此任務的所有歷史紀錄。`)) window.deleteChoreFromCloud(choreId); };

        window.openPinModal = (user, msg) => { document.getElementById('pinPromptText').innerText = msg; document.getElementById('pinAvatarDisplay').innerText = user.icon; document.getElementById('verifyPin').value = ''; document.getElementById('pinModal').style.display = 'flex'; document.getElementById('verifyPin').focus(); };

        window.submitPin = () => {

            if(!pendingAction) return; const input = document.getElementById('verifyPin').value;

            if(input === pendingAction.correctPin) { if(pendingAction.type === 'deleteRecord') window.updateRecordInCloud(pendingAction.choreId, pendingAction.dateStr, pendingAction.slotIndex, null); else if (pendingAction.type === 'deleteUser') window.deleteUserFromCloud(pendingAction.userId); window.closeModal('pinModal'); } else { alert("密碼錯誤"); document.getElementById('verifyPin').value = ''; }

        };

        window.selectChoreMode = (mode) => { tempMode = mode; const opts = document.querySelectorAll('#choreModeSelect .radio-option'); opts.forEach(o => o.classList.remove('selected')); if(mode === 'single') opts[0].classList.add('selected'); else opts[1].classList.add('selected'); };

        window.closeModal = (id) => { document.getElementById(id).style.display = 'none'; pendingAction = null; };

        function bindLongPress(element, callback) { const start = (e) => { pressTimer = setTimeout(() => { if(navigator.vibrate) navigator.vibrate(50); callback(); }, 800); }; const end = () => clearTimeout(pressTimer); element.addEventListener('mousedown', start); element.addEventListener('touchstart', start); element.addEventListener('mouseup', end); element.addEventListener('mouseleave', end); element.addEventListener('touchend', end); }

        function renderIconGrid(divId, list) { const div = document.getElementById(divId); div.innerHTML = ''; list.forEach(ic => { const el = document.createElement('div'); el.className = `icon-option ${ic === tempIcon ? 'selected' : ''}`; el.innerText = ic; el.onclick = () => { tempIcon = ic; renderIconGrid(divId, list); }; div.appendChild(el); }); }
